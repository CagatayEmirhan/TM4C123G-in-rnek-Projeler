void Kesme()
{
    TimerIntClear(TIMER0_BASE, TIMER_TIMA_TIMEOUT);
    timerACounter++;
    kursor_basa_don();
    Saniye_1();
}

void Saniye_1()
{
    if(timerACounter==2000)
    {
        saniye_1++;
        lcd_write(saniye_1);
        SysCtlDelay(2000);
        kursor_sola_kaydır();
        timerACounter=0;
        if(saniye_1==10)
        {
            saniye_1=0;
            lcd_write(saniye_1);
            SysCtlDelay(2000);
            kursor_sola_kaydır();
            SysCtlDelay(2000);
            kursor_sola_kaydır();
            SysCtlDelay(2000);
            a++;
            SysCtlDelay(2000);
            timerACounter=0;
            saniye_2++;
            Saniye_2();
        }
    }
}

void lcd_write(unsigned char d )
{
    d=d+(0x30);
    GPIOPinWrite(LCD_PORT, RS, 0x01);
    SysCtlDelay(2000);
    GPIOPinWrite(LCD_PORT, E, 0x02);
    SysCtlDelay(2000);
    GPIOPinWrite(LCD_PORT, D4 | D5 | D6 | D7, (d & 0xf0) );
    SysCtlDelay(2000);
    GPIOPinWrite(LCD_PORT, E, 0x00);

    SysCtlDelay(50000);

    GPIOPinWrite(LCD_PORT, D4 | D5 | D6 | D7, (d & 0x0f) << 4 );
    SysCtlDelay(2000);
    GPIOPinWrite(LCD_PORT, RS, 0x01);
    SysCtlDelay(2000);
    GPIOPinWrite(LCD_PORT, E, 0x02);
    SysCtlDelay(2000);
    GPIOPinWrite(LCD_PORT, E, 0x00);

    SysCtlDelay(50000);
}

void ilk_ayar()
{
    // Sistem saati 40 MHz olarak ayarlanıyor.
    SysCtlClockSet(SYSCTL_SYSDIV_5 | SYSCTL_USE_PLL | SYSCTL_XTAL_16MHZ | SYSCTL_OSC_MAIN);
    SysCtlPeripheralEnable(SYSCTL_PERIPH_GPIOB);
    GPIOPinTypeGPIOOutput(GPIO_PORTB_BASE,GPIO_PIN_0|GPIO_PIN_1|GPIO_PIN_4|GPIO_PIN_5|GPIO_PIN_6|GPIO_PIN_7);
    // Timer0 etkinleştiriliyor.
    SysCtlPeripheralEnable(SYSCTL_PERIPH_TIMER0);
    TimerConfigure(TIMER0_BASE, TIMER_CFG_SPLIT_PAIR | TIMER_CFG_A_PERIODIC | TIMER_CFG_B_PERIODIC); // Timer0 A ve B ayrı olarak periyodik modda ayarlanıyor.
    // TimerA yük değeri ayarlanıyor (???). 1 sn de bir kesme olacak kırmızı ledi terslicek
    TimerLoadSet(TIMER0_BASE, TIMER_A, (12000*12)); // 16 bit limit içinde ??? olarak ayarlandı
    // Kesme ayarları - global ve yerel interruptlar etkinleştiriliyor.
    IntMasterEnable(); // Global kesme izni
    IntEnable(INT_TIMER0A); // Timer0A için kesme izni
    // TimerA ve TimerB kesme ayarları.
    TimerIntEnable(TIMER0_BASE, TIMER_TIMA_TIMEOUT); // TimerA kesmesi etkinleştiriliyor.
    // TimerA ve TimerB kesme fonksiyonları kaydediliyor.
    TimerIntRegister(TIMER0_BASE, TIMER_A, Dakika_1);
    // TimerA ve TimerB başlatılıyor.
    TimerEnable(TIMER0_BASE, TIMER_A);
}
